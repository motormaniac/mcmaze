# Maze Bounds Settings
-- Set default settings
scoreboard players set .temp x 9
scoreboard players set .temp z 9
scoreboard players set .temp level 2
scoreboard players set .temp multiblockPercentage 20

-- Each settings area has 1. increase, 2. decrease, 3. update sign
-- update sign should be chained after both increase and decrease. Modify the y coordinate accordingly (+1 or -1)
## x  (bounds: 5..15)
execute if score .temp x matches 5..14 run scoreboard players add .temp x 1
data merge block ~3 ~-1 ~ {front_text:{messages:["width:",{"score":{"name":".temp","objective":"x"}},"",""]}}
execute if score .temp x matches 6..15 run scoreboard players remove .temp x 1
data merge block ~3 ~1 ~ {front_text:{messages:["width:",{"score":{"name":".temp","objective":"x"}},"",""]}}

## z  (bounds: 5..15)
execute if score .temp z matches 5..14 run scoreboard players add .temp z 1
data merge block ~3 ~-1 ~ {front_text:{messages:["length:",{"score":{"name":".temp","objective":"z"}},"",""]}}
execute if score .temp z matches 6..15 run scoreboard players remove .temp z 1
data merge block ~3 ~1 ~ {front_text:{messages:["length:",{"score":{"name":".temp","objective":"z"}},"",""]}}

## level  (bounds: 1..10)
execute if score .temp level matches 1..9 run scoreboard players add .temp level 1
data merge block ~3 ~-1 ~ {front_text:{messages:["height:",{"score":{"name":".temp","objective":"level"}},"",""]}}
execute if score .temp level matches 2..10 run scoreboard players remove .temp level 1
data merge block ~3 ~1 ~ {front_text:{messages:["height:",{"score":{"name":".temp","objective":"level"}},"",""]}}

## multiblockPercentage  (bounds: 10..50)
execute if score .temp multiblockPercentage matches 10..49 run scoreboard players add .temp multiblockPercentage 5
data merge block ~3 ~-1 ~ {front_text:{messages:["percentage:",{"score":{"name":".temp","objective":"multiblockPercentage"}},"",""]}}
execute if score .temp multiblockPercentage matches 11..50 run scoreboard players remove .temp multiblockPercentage 5
data merge block ~3 ~1 ~ {front_text:{messages:["percentage:",{"score":{"name":".temp","objective":"multiblockPercentage"}},"",""]}}
